# OrderFlow - Python packages
#
# Install with:
#     py -m pip install -r requirements.txt
#
# pywin32 provides both win32com and pythoncom, which OrderFlow uses to
# drive Outlook (reading mail) and Excel (reading DRM-protected files).
# Missing it is what produces "No module named 'win32com'".

pywin32                  # Outlook + Excel COM automation
openpyxl                 # read/write .xlsx (ERP template, price sheets)
